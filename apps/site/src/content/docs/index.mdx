import { Card, CardGrid } from "@astrojs/starlight/components";

<CardGrid>
  <Card title="Quick Share" icon="rocket">
    Pick a port, get a `trycloudflare.com` URL. No account needed. Stop it whenever you
    like, or set it to stop on its own.
  </Card>
  <Card title="Your own domains" icon="seti:config">
    Route `app.example.com` to `localhost:3000`. Teitunnel creates the tunnel and the
    DNS record, shows you every change before it makes it, and checks the result.
  </Card>
  <Card title="Safe by design" icon="approve-check">
    Tokens live in the macOS keychain. Every Cloudflare change is previewed, logged and
    undone if a step fails. DNS records Teitunnel didn't create are never deleted
    without asking.
  </Card>
  <Card title="Always on" icon="setting">
    Keep routes running after you quit and across restarts. Traffic, logs and the
    Doctor tell you what's happening and how to fix it.
  </Card>
</CardGrid>
